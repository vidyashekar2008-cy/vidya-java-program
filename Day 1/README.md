public class EvenOdd {
    public static void main(String[] args) {

        int a = 24;
        int b = 37;

        if (a % 2 == 0) {
            System.out.println(a + " is Even");
        } else {
            System.out.println(a + " is Odd");
        }

        if (b % 2 == 0) {
            System.out.println(b + " is Even");
        } else {
            System.out.println(b + " is Odd");
        }
    }
}
